async function loadScheduledCampaignsList() { const container = document.getElementById('scheduledCampaignsContainer'); if (!SESSION_ID || SESSION_ID === 'unpaired_user') { container.innerHTML = '
Not connected.
'; return; } try { const res = await fetch(`${FIXED_BACKEND_URL}/api/campaign/scheduled-jobs/${SESSION_ID}`, { headers: { 'Authorization': `Bearer ${API_BEARER_TOKEN}`, 'ngrok-skip-browser-warning': 'true' } }); const d = await res.json(); if (d.success && Array.isArray(d.jobs) && d.jobs.length > 0) { container.innerHTML = d.jobs.map(j => `
⏰ ${escapeHtml(j.scheduledForDate)}
${j.recipientCount} recipients • ${j.mediaCount} media items
`).join(''); } else { container.innerHTML = '
No upcoming scheduled campaigns. (Max: 3)
'; } } catch (e) { container.innerHTML = ''; } }
